# 53. Maximum Subarray

Given an integer array nums, find the subarray with the largest sum, and return its sum.

 

Example 1:

Input: nums = [-2,1,-3,4,-1,2,1,-5,4]
Output: 6
Explanation: The subarray [4,-1,2,1] has the largest sum 6.
Example 2:

Input: nums = [1]
Output: 1
Explanation: The subarray [1] has the largest sum 1.
Example 3:

Input: nums = [5,4,-1,7,8]
Output: 23
Explanation: The subarray [5,4,-1,7,8] has the largest sum 23.
 

Constraints:

1 <= nums.length <= 105
-104 <= nums[i] <= 104
 

Follow up: If you have figured out the O(n) solution, try coding another solution using the divide and conquer approach, which is more subtle.

- Dynamic Programming의 기초를 다지기 위한 아주 훌륭한 문제 

동적 계획법의 핵심은 "큰 문제를 작은 문제로 쪼개고, 과거의 정답(상태)을 재활용하여 현재의 정답을 구하는 것"입니다.

작성하신 코드에서 그 역할을 하는 것이 바로 `prev` 변수입니다.
일반적인 교과서에서는 배열을 만들어 `dp[i]`에 값을 저장하지만, 질문자님께서는 직전의 상태만 필요하다는 것을 간파하고 `prev`라는 단일 변수로 메모리 공간을 $O(1)$로 최적화하셨습니다.

질문자님의 로직을 점화식(수식)으로 표현하면 다음과 같습니다.


$$DP[i] = \max(DP[i-1] + nums[i], nums[i])$$

작성하신 `if-else` 문은 위 점화식을 그대로 논리적으로 풀어쓴 것입니다.

### 2. 시각적 그림으로 보는 알고리즘 동작 원리

초보자들도 이해하기 쉽도록 질문자님의 `if-else` 로직을 시각화해 보겠습니다.
배열 `nums = [-2, 1, -3, 4]` 가 있다고 가정해 봅시다.

```text
[초기 상태]
배열 (nums):  -2      1     -3      4
            ---
prev      :  -2  (첫 번째 값으로 초기화)
max_value :  -2

[i = 1 순회] 현재 숫자: 1
1. temp = prev(-2) + 현재 숫자(1) = -1
2. temp(-1) 과 현재 숫자(1) 비교!
   => "과거의 유산(-2)을 안고 갔더니(-1) 오히려 내 자신(1)보다 작아졌네?"
   => "과거를 버리고 나(1)부터 새로 시작하자!" (else 구문 실행)
3. prev = 1
4. max_value = std::max(-2, 1) = 1

[i = 2 순회] 현재 숫자: -3
1. temp = prev(1) + 현재 숫자(-3) = -2
2. temp(-2) 과 현재 숫자(-3) 비교!
   => "과거의 유산(1)을 안고 갔더니(-2), 그래도 내 자신(-3)보다는 크네!"
   => "과거를 이어가자!" (if 구문 실행)
3. prev = -2
4. max_value = std::max(1, -2) = 1  (최대값 유지)

[i = 3 순회] 현재 숫자: 4
1. temp = prev(-2) + 현재 숫자(4) = 2
2. temp(2) 과 현재 숫자(4) 비교!
   => "과거의 유산(-2)을 안고 갔더니(2), 내 자신(4)보다 작아졌네?"
   => "버리고 4부터 새로 시작하자!" (else 구문 실행)
3. prev = 4
4. max_value = std::max(1, 4) = 4  (최대값 갱신!)

==> 최종 정답: 4

```

### 3. 라인 단위 코드 해설 (C++ 기준)

질문자님의 코드를 기반으로, 면접이나 실무에서 사용하기 좋게 주석을 달아 라인 단위로 설명해 드립니다.

```cpp
// solution.cpp
#include <iostream>
#include <vector>
#include <algorithm> // std::max 를 사용하기 위한 표준 라이브러리

using namespace std;

class Solution {
public:
    int maxSubArray(vector<int>& nums) {
        // [1] 초기화: 첫 번째 원소를 글로벌 최대값과 이전 DP 상태값으로 설정합니다.
        int max_value = nums[0]; 
        int prev = nums[0];      
        
        // [2] 1번 인덱스부터 배열의 끝까지 순회합니다.
        for(int i = 1; i < nums.size(); i++){
            // [3] temp는 '이전까지의 연속된 합(prev)'에 '현재 숫자(nums[i])'를 더한 값입니다.
            auto temp = prev + nums[i];
            
            // [4] 동적 계획법의 핵심 상태 전이 조건문입니다.
            if( temp > nums[i] ){
                // (이전 합 + 현재 숫자)가 (현재 숫자 단독)보다 크다면, 
                // 이전부터 이어져 온 배열에 현재 숫자를 붙이는 것이 이득입니다.
                prev = temp;
            } else {
                // 반대로 (현재 숫자 단독)이 더 크다면, (즉, prev가 음수여서 갉아먹었다면)
                // 과거를 끊어내고 현재 숫자부터 새로운 부분 배열을 시작합니다.
                prev = nums[i];
            }
            
            // [5] std::max(a, b) 표준 라이브러리 함수
            // a와 b 중 더 큰 값을 반환합니다.
            // 매 단계마다 갱신된 로컬 최대합(prev)과 전체 글로벌 최대합(max_value)을 비교해 갱신합니다.
            max_value = std::max(max_value, prev);
        }
        
        // [6] 배열을 모두 순회한 후 가장 컸던 값을 반환합니다.
        return max_value;
    }
};

int main() {
    Solution sol;
    vector<int> nums = {-2, 1, -3, 4, -1, 2, 1, -5, 4};
    cout << "최대 부분 배열의 합: " << sol.maxSubArray(nums) << endl; 
    // 출력: 6
    return 0;
}

```

### 4. 다른 언어에서의 활용 (Python & C#)

알고리즘 전문가로서, 질문자님의 훌륭한 로직을 파이썬과 C# 스크립트로도 동일하게 구현해 드립니다. 다양한 환경에서도 동일한 DP 사고방식이 적용됨을 확인해 보세요.

**Python (`solution.py`)**

```python
# 파이썬은 내장 함수인 max()를 사용하여 if-else 문을 더욱 간결하게 표현할 수 있습니다.
from typing import List

class Solution:
    def maxSubArray(self, nums: List[int]) -> int:
        max_value = nums[0]
        prev = nums[0]
        
        # 인덱스 1부터 끝까지 순회
        for i in range(1, len(nums)):
            # 질문자님의 if-else 로직은 파이썬 내장 함수 max() 하나로 완벽히 치환됩니다.
            # "이전까지의 합+현재숫자(prev + nums[i])" 와 "현재숫자(nums[i])" 중 큰 것을 prev로 갱신
            prev = max(prev + nums[i], nums[i])
            
            # 전체 최대값 갱신
            max_value = max(max_value, prev)
            
        return max_value

# 실행부
if __name__ == "__main__":
    sol = Solution()
    print("최대 부분 배열의 합:", sol.maxSubArray([-2, 1, -3, 4, -1, 2, 1, -5, 4]))

```

**C# (`solution.csx`)**

```csharp
// C# 스크립트(.csx) 환경에서 바로 실행 가능한 코드입니다.
using System;

public class Solution {
    public int MaxSubArray(int[] nums) {
        int max_value = nums[0];
        int prev = nums[0];
        
        for(int i = 1; i < nums.Length; i++) {
            int temp = prev + nums[i];
            
            // 질문자님의 원본 로직을 그대로 C# 문법에 맞게 적용
            if(temp > nums[i]) {
                prev = temp;
            } else {
                prev = nums[i];
            }
            
            // System.Math.Max 를 사용하여 최대값을 갱신합니다.
            max_value = Math.Max(max_value, prev);
        }
        
        return max_value;
    }
}

// 실행부
Solution sol = new Solution();
int[] nums = new int[] {-2, 1, -3, 4, -1, 2, 1, -5, 4};
Console.WriteLine($"최대 부분 배열의 합: {sol.MaxSubArray(nums)}");
```
