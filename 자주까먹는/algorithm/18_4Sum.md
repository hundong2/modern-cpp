안녕하십니까, 코드 알고리즘 최고 박사 이동훈입니다. 제시해주신 '4Sum' 문제에 대한 명쾌하고 깊이 있는 아키텍처 설계를 시작하겠습니다.

이 문제는 $O(N^4)$의 완전 탐색(Brute-force)으로 접근하면 'Time Limit Exceeded'가 발생합니다. 따라서 배열을 미리 정렬한 뒤, 두 개의 숫자를 고정하고 나머지 두 숫자를 **투 포인터(Two Pointers)** 기법으로 탐색하여 시간 복잡도를 $O(N^3)$으로 단축하는 것이 핵심입니다.

---

## 1. 알고리즘 시각화 및 직관적 이해

이해를 돕기 위해 정렬된 배열에서 포인터가 어떻게 움직이는지 도식화해 보겠습니다.

```text
[ 초기 상태: nums = {1, 0, -1, 0, -2, 2}, Target = 0 ]

1단계: 표준 라이브러리를 활용한 오름차순 정렬
[ -2, -1, 0, 0, 1, 2 ]

2단계: 첫 번째 숫자(i)와 두 번째 숫자(j) 고정 후 양끝 포인터(left, right) 설정
  i   j  left       right
[ -2, -1, 0,  0,  1,  2 ] 
  ^   ^   ^           ^

3단계: 네 숫자의 합 계산 및 포인터 이동
합 = (-2) + (-1) + 0 + 2 = -1
- 합(-1) < Target(0) 이므로, 값을 키우기 위해 left 포인터를 우측으로 이동합니다.

```

중복된 값을 건너뛰는 로직(Skip Duplicates)을 완벽히 구현해야만 문제의 요구사항인 "중복 없는 고유한 배열"을 반환할 수 있습니다.

## 2. C++ 구현 (cpp)

C++의 `<algorithm>` 헤더에 포함된 `std::sort`를 적극 활용하며, 오버플로우를 방지하기 위해 64비트 정수형(`long long`) 캐스팅을 적용합니다.

```cpp
#include <vector>
#include <algorithm> // std::sort 사용을 위한 헤더

using namespace std;

class Solution {
public:
    vector<vector<int>> fourSum(vector<int>& nums, int target) {
        vector<vector<int>> result;
        int n = nums.size();
        
        // 원소 개수가 4개 미만이면 조건을 만족할 수 없으므로 빈 배열 반환
        if (n < 4) return result; 
        
        // std::sort: 평균 시간복잡도 O(N log N)의 최적화된 정렬 알고리즘(IntroSort)
        sort(nums.begin(), nums.end()); 
        
        for (int i = 0; i < n - 3; ++i) {
            // 첫 번째 원소 중복 건너뛰기
            if (i > 0 && nums[i] == nums[i - 1]) continue; 
            
            for (int j = i + 1; j < n - 2; ++j) {
                // 두 번째 원소 중복 건너뛰기
                if (j > i + 1 && nums[j] == nums[j - 1]) continue; 
                
                int left = j + 1;       // 세 번째 숫자를 가리키는 포인터
                int right = n - 1;      // 네 번째 숫자를 가리키는 포인터
                
                while (left < right) {
                    // 합계 과정에서 int 오버플로우를 막기 위해 long long으로 형변환
                    long long sum = (long long)nums[i] + nums[j] + nums[left] + nums[right];
                    
                    if (sum == target) {
                        // 정답을 찾은 경우 결과 배열에 추가
                        result.push_back({nums[i], nums[j], nums[left], nums[right]});
                        
                        // 세 번째, 네 번째 원소의 중복을 모두 건너뜀
                        while (left < right && nums[left] == nums[left + 1]) left++;
                        while (left < right && nums[right] == nums[right - 1]) right--;
                        
                        left++;  // 다음 탐색을 위해 포인터 이동
                        right--; // 다음 탐색을 위해 포인터 이동
                    } 
                    else if (sum < target) {
                        left++;  // 합이 작으면 값을 키우기 위해 left를 우측으로
                    } 
                    else {
                        right--; // 합이 크면 값을 줄이기 위해 right를 좌측으로
                    }
                }
            }
        }
        return result;
    }
};

```

## 3. Python 구현 (py)

Python의 내장 `sort()` 메서드는 팀소트(Timsort) 알고리즘을 사용하며 매우 효율적입니다. 파이썬은 정수 오버플로우를 자체적으로 처리하므로 타입 캐스팅이 필요 없습니다.

```python
class Solution:
    def fourSum(self, nums: list[int], target: int) -> list[list[int]]:
        n = len(nums)
        result = []
        
        # 리스트가 4개 미만이면 탐색 불가
        if n < 4:
            return result
            
        # list.sort(): 원본 리스트를 제자리에서(in-place) 정렬하는 표준 함수
        nums.sort()
        
        for i in range(n - 3):
            # 첫 번째 숫자의 중복 방지
            if i > 0 and nums[i] == nums[i - 1]:
                continue
                
            for j in range(i + 1, n - 2):
                # 두 번째 숫자의 중복 방지
                if j > i + 1 and nums[j] == nums[j - 1]:
                    continue
                    
                left = j + 1
                right = n - 1
                
                while left < right:
                    # 네 숫자의 합 계산
                    total = nums[i] + nums[j] + nums[left] + nums[right]
                    
                    if total == target:
                        # 리스트의 append() 메서드로 찾은 조합을 결과에 추가
                        result.append([nums[i], nums[j], nums[left], nums[right]])
                        
                        # 중복 원소를 스킵하기 위한 while 루프
                        while left < right and nums[left] == nums[left + 1]:
                            left += 1
                        while left < right and nums[right] == nums[right - 1]:
                            right -= 1
                            
                        # 양방향 포인터 모두 한 칸씩 이동하여 새로운 조합 탐색
                        left += 1
                        right -= 1
                    elif total < target:
                        left += 1  # 타겟보다 작으므로 하한선 올리기
                    else:
                        right -= 1 # 타겟보다 크므로 상한선 내리기
                        
        return result

```

## 4. C# 구현 (csx)

C#에서는 `Array.Sort()`를 사용하여 정렬하며, C++과 마찬가지로 32비트 `int`의 한계를 초과하는 테스트케이스에 대비해 `long` 타입 캐스팅이 필수적입니다.

```csharp
using System;
using System.Collections.Generic;

public class Solution {
    public IList<IList<int>> FourSum(int[] nums, int target) {
        IList<IList<int>> result = new List<IList<int>>();
        int n = nums.Length;
        
        if (n < 4) return result;
        
        // Array.Sort(): 배열을 오름차순으로 정렬하는 닷넷 프레임워크 표준 메서드
        Array.Sort(nums);
        
        for (int i = 0; i < n - 3; i++) {
            // 중복된 i 값 스킵
            if (i > 0 && nums[i] == nums[i - 1]) continue;
            
            for (int j = i + 1; j < n - 2; j++) {
                // 중복된 j 값 스킵
                if (j > i + 1 && nums[j] == nums[j - 1]) continue;
                
                int left = j + 1;
                int right = n - 1;
                
                while (left < right) {
                    // C#에서 int 오버플로우 방지를 위한 (long) 명시적 형변환
                    long sum = (long)nums[i] + nums[j] + nums[left] + nums[right];
                    
                    if (sum == target) {
                        // List를 생성하여 결과 컬렉션에 추가
                        result.Add(new List<int> { nums[i], nums[j], nums[left], nums[right] });
                        
                        // left와 right 포인터의 중복 값 회피
                        while (left < right && nums[left] == nums[left + 1]) left++;
                        while (left < right && nums[right] == nums[right - 1]) right--;
                        
                        left++;
                        right--;
                    } 
                    else if (sum < target) {
                        left++;
                    } 
                    else {
                        right--;
                    }
                }
            }
        }
        return result;
    }
}

```
