# 40. Combination Sum II

Given a collection of candidate numbers (candidates) and a target number (target), find all unique combinations in candidates where the candidate numbers sum to target.

Each number in candidates may only be used once in the combination.

Note: The solution set must not contain duplicate combinations.

 

Example 1:

Input: candidates = [10,1,2,7,6,1,5], target = 8
Output: 
[
[1,1,6],
[1,2,5],
[1,7],
[2,6]
]
Example 2:

Input: candidates = [2,5,2,1,2], target = 5
Output: 
[
[1,2,2],
[5]
]

## My Solution

### C++

```cpp
class Solution {
public:
    vector<vector<int>> combinationSum2(vector<int>& candidates, int target) {
        vector<vector<int>> result_;
        vector<int> temp;
        std::sort(candidates.begin(), candidates.end());
        seek(result_, temp, candidates, target, 0);
        return result_;
    }
private: 
    void seek(vector<vector<int>>& result, vector<int> temp, vector<int>& candidates, int target, int idx){
        if( target == 0 ){
            result.push_back(temp);
            return;
        } else if ( target < 0 ){
            return;
        }
        for(int i = idx; i < candidates.size(); i++){
            if( i > idx && candidates[i] == candidates[i-1] ) continue;
            if( target - candidates[i] < 0 ) return;
            temp.push_back(candidates[i]);
            seek(result, temp, candidates, target - candidates[i], i + 1);
            temp.pop_back();
        }
    }
};
```

public class Solution {
    public IList<IList<int>> CombinationSum2(int[] candidates, int target) {
        Array.Sort(candidates);
        List<IList<int>> result = new();
        List<int> current = new();
        Seek(ref result, current, ref candidates, target, 0);
        return result;

    }
    public void Seek(ref List<IList<int>> result, List<int> current, ref int[] candidates, int target, int idx){
        if( target == 0 ){
            result.Add(new List<int>(current));
            return;
        } else if( target < 0){
            return;
        }
        for(int i = idx; i < candidates.Count(); i++){
            if( target - candidates[i] < 0 ) return;
            if (i > idx && candidates[i] == candidates[i - 1]) continue;
            current.Add(candidates[i]);
            Seek(ref result, current, ref candidates, target - candidates[i], i + 1);
            current.RemoveAt(current.Count()-1);
        }
    }
}
