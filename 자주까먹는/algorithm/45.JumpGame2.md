# 45. Jump Game II

You are given a 0-indexed array of integers nums of length n. You are initially positioned at index 0.

Each element nums[i] represents the maximum length of a forward jump from index i. In other words, if you are at index i, you can jump to any index (i + j) where:

0 <= j <= nums[i] and
i + j < n
Return the minimum number of jumps to reach index n - 1. The test cases are generated such that you can reach index n - 1.

 

Example 1:

Input: nums = [2,3,1,1,4]
Output: 2
Explanation: The minimum number of jumps to reach the last index is 2. Jump 1 step from index 0 to 1, then 3 steps to the last index.
Example 2:

Input: nums = [2,3,0,1,4]
Output: 2
 

Constraints:

1 <= nums.length <= 104
0 <= nums[i] <= 1000
It's guaranteed that you can reach nums[n - 1].

## My Solution

### cpp

```cpp
class Solution {
public:
    int jump(vector<int>& nums) {
        int n = nums.size();
        if(n <= 1) return 0;
        int nextLevel = 0;
        int currentLevel = 0;
        int count = 0;
        for(int i = 0; i < n; i++){
            nextLevel = std::max(nextLevel, i + nums[i]);
            if( currentLevel == i){
                currentLevel = nextLevel;
                count++;
                if( currentLevel >= n - 1){
                    break;
                }
            }
        }
        return count;
    }
};
```

```csharp
public class Solution {

        int curr = 0;
        int maxNext = 0;
        int cnt = 0;

            maxNext = Math.Max(maxNext, i + nums[i]);

                cnt++;
                curr = maxNext;
                if( curr >= nums.Length - 1){
                    break;
                }
            }
        }
        return cnt;
    }

}
```

**탐욕법(Greedy Algorithm) 기반 최소 점프 탐색**

이 문제는 현재 위치에서 도달할 수 있는 가장 먼 거리를 지속적으로 갱신하며, 현재 점프의 한계 지점(경계)에 도달했을 때만 점프 횟수를 증가시키는 방식으로 최적화합니다. 시간 복잡도 O(N) 및 공간 복잡도 O(1)로 해결할 수 있습니다.

**작동 원리 시각화 (예제: `nums = [2, 3, 1, 1, 4]`)**

배열을 순회하면서 세 가지 핵심 변수를 관리합니다.

* `jumps`: 현재까지의 총 점프 횟수
* `current_end`: 현재 점프 횟수로 도달할 수 있는 최대 인덱스 경계
* `farthest`: 지금까지 탐색한 위치들에서 도달 가능한 가장 먼 인덱스

```text
인덱스(i):    0      1      2      3      4
배열(nums): [ 2,     3,     1,     1,     4 ]
           |----|----|
             점프 1 (최대 도달 경계 current_end = 2)
           
                  |----|----|----|
                    점프 2 (새로운 경계 current_end = 4)

[순회 과정]
i=0: nums[0]=2 ➔ 가장 먼 곳(farthest)은 0+2=2. 
     현재 인덱스가 current_end(0)이므로 jumps=1, current_end=2로 갱신.
i=1: nums[1]=3 ➔ farthest는 max(2, 1+3) = 4. (아직 current_end 2에 도달 안 함)
i=2: nums[2]=1 ➔ farthest는 max(4, 2+1) = 4.
     현재 인덱스가 current_end(2)이므로 jumps=2, current_end=4로 갱신.
i=3: nums[3]=1 ➔ farthest는 max(4, 3+1) = 4.
i=4: 마지막 인덱스 도달 (종료). 총 점프 횟수: 2

```

**Python 구현 및 라인별 설명 (`jump_game.py`)**

```python
from typing import List

class Solution:
    def jump(self, nums: List[int]) -> int:
        # 배열의 길이가 1 이하인 경우 점프가 필요 없으므로 0 반환
        if len(nums) <= 1:
            return 0
        
        jumps = 0        # 총 점프 횟수를 저장하는 변수
        current_end = 0  # 현재 점프 횟수로 갈 수 있는 최대 경계선
        farthest = 0     # 다음 점프를 위해 탐색한 곳 중 가장 먼 인덱스
        
        # 마지막 인덱스 직전까지만 순회 (마지막 인덱스에 도달하면 더 이상 점프할 필요 없음)
        for i in range(len(nums) - 1):
            # 1. 현재 위치 i에서 점프할 수 있는 가장 먼 거리를 갱신
            # 표준 내장 함수인 max를 적극 활용하여 최댓값 유지
            farthest = max(farthest, i + nums[i])
            
            # 2. 현재 탐색 위치가 현재 점프의 경계선에 도달한 경우
            if i == current_end:
                jumps += 1              # 경계를 넘어야 하므로 점프 횟수 1 증가
                current_end = farthest  # 다음 점프의 경계선을 가장 멀리 갈 수 있었던 곳으로 갱신
                
                # 만약 새 경계선이 배열의 끝이거나 그 이상이라면, 더 탐색할 필요 없이 종료 가능
                if current_end >= len(nums) - 1:
                    break
                    
        return jumps

```

**C++ 구현 및 라인별 설명 (`jump_game.cpp`)**

```cpp
#include <iostream>
#include <vector>
#include <algorithm> // std::max 사용을 위해 포함

class Solution {
public:
    int jump(std::vector<int>& nums) {
        int n = nums.size();
        // 예외 처리: 목적지가 이미 현재 위치인 경우
        if (n <= 1) return 0;

        int jumps = 0;       // 최소 점프 횟수 기록
        int current_end = 0; // 현재 점프 구간의 끝 인덱스
        int farthest = 0;    // 도달 가능한 최대 인덱스

        // n - 1 까지만 반복 (마지막 요소에서는 점프할 필요가 없으므로 제외)
        for (int i = 0; i < n - 1; ++i) {
            // std::max 표준 라이브러리 함수를 사용하여 최장 도달 거리 지속 업데이트
            farthest = std::max(farthest, i + nums[i]);

            // 현재 점프 구간의 끝에 도달했을 때
            if (i == current_end) {
                jumps++;             // 실제로 점프를 수행
                current_end = farthest; // 새로운 점프 구간 설정
                
                // 조기 종료 조건: 갱신된 구간이 이미 목표 지점을 덮은 경우
                if (current_end >= n - 1) break;
            }
        }
        
        return jumps;
    }
};

// 실행을 위한 테스트 뼈대 코드
int main() {
    Solution sol;
    std::vector<int> nums = {2, 3, 1, 1, 4};
    std::cout << "Minimum jumps: " << sol.jump(nums) << std::endl; // 출력: 2
    return 0;
}

```

**C# 구현 및 라인별 설명 (`jump_game.csx`)**

```csharp
using System;
// C# 스크립트(csx) 환경에서 즉시 실행 가능한 형태

public class Solution {
    public int Jump(int[] nums) {
        int n = nums.Length;
        // 배열 길이가 1이면 점프 불필요
        if (n <= 1) return 0;

        int jumps = 0;
        int currentEnd = 0;
        int farthest = 0;

        // 마지막 요소 도달 시점에서 점프를 발생시키지 않기 위해 n-1 미만으로 순회
        for (int i = 0; i < n - 1; i++) {
            // System.Math.Max 를 통해 현재 위치에서 도달 가능한 최대 거리 비교
            farthest = Math.Max(farthest, i + nums[i]);

            // 현재 스텝의 한계 지점에 도달했는지 확인
            if (i == currentEnd) {
                jumps++;                 // 점프 카운트 증가
                currentEnd = farthest;   // 경계를 다음 점프의 최대 도달 거리로 확장
                
                // 최적화: 이미 끝에 도달할 수 있는 경우 불필요한 연산 방지
                if (currentEnd >= n - 1) {
                    break;
                }
            }
        }

        return jumps;
    }
}

// 테스트 실행부
Solution sol = new Solution();
int[] nums = { 2, 3, 1, 1, 4 };
Console.WriteLine($"Minimum jumps: {sol.Jump(nums)}");
```
