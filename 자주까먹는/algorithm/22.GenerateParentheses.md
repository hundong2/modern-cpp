## 22. Generate Parentheses

Given n pairs of parentheses, write a function to generate all combinations of well-formed parentheses.

 

Example 1:

Input: n = 3
Output: ["((()))","(()())","(())()","()(())","()()()"]
Example 2:

Input: n = 1
Output: ["()"]
 

Constraints:

1 <= n <= 8

### My Solution

```csharp
public class Solution {
    public IList<string> GenerateParenthesis(int n) {
        List<string> result = new();
        StringBuilder value = new();
        LoopCheck(0, 0, n, value, result);
        return result;
    }
    private void LoopCheck(int st, int ed, int max, StringBuilder value, IList<string> result)
    {
        if( st == max && ed == max)
        {
            result.Add(value.ToString());
        }
        if( st < max )
        {
            value.Append('(');
            LoopCheck(st + 1, ed, max, value,  result);
            value.Length--;
        }
        if( st > ed)
        {
            value.Append(')');
            LoopCheck(st, ed + 1, max, value, result);
            value.Length--;
        }
    }
}
```

```cpp
class Solution {
public:
    vector<string> generateParenthesis(int n) {
        std::vector<std::string> result {};
        std::string v{};
        depth(0 , 0, n, v, result);
        return result;
    }
private: 
    void depth(const int& st_cnt, const int& cls_cnt, int max, string& value, std::vector<std::string>& re) {
        if( st_cnt == max && cls_cnt == max ){
            //check
            re.push_back(value);
        } else {
            if( st_cnt < max) {
                value.push_back('(');
                depth(st_cnt + 1, cls_cnt, max, value, re);
                value.pop_back();
            }
                
            if(st_cnt > cls_cnt) {
                value.push_back(')');
                depth(st_cnt, cls_cnt + 1, max, value, re);
                value.pop_back();
            }
                
        }
    }
    
};
```
