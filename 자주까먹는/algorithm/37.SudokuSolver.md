# 37. Sudoku Solver

## Hard

Hint
Write a program to solve a Sudoku puzzle by filling the empty cells.

A sudoku solution must satisfy all of the following rules:

Each of the digits 1-9 must occur exactly once in each row.
Each of the digits 1-9 must occur exactly once in each column.
Each of the digits 1-9 must occur exactly once in each of the 9 3x3 sub-boxes of the grid.
The '.' character indicates empty cells.

 

Example 1:


Input: board = [["5","3",".",".","7",".",".",".","."],["6",".",".","1","9","5",".",".","."],[".","9","8",".",".",".",".","6","."],["8",".",".",".","6",".",".",".","3"],["4",".",".","8",".","3",".",".","1"],["7",".",".",".","2",".",".",".","6"],[".","6",".",".",".",".","2","8","."],[".",".",".","4","1","9",".",".","5"],[".",".",".",".","8",".",".","7","9"]]
Output: [["5","3","4","6","7","8","9","1","2"],["6","7","2","1","9","5","3","4","8"],["1","9","8","3","4","2","5","6","7"],["8","5","9","7","6","1","4","2","3"],["4","2","6","8","5","3","7","9","1"],["7","1","3","9","2","4","8","5","6"],["9","6","1","5","3","7","2","8","4"],["2","8","7","4","1","9","6","3","5"],["3","4","5","2","8","6","1","7","9"]]

board[i][j] is a digit or '.'.

class Solution {
public:
    bool solveSudoku(vector<vector<char>>& board) {
        for(int r = 0; r < 9; r++){
            for(int c = 0; c < 9; c++){
                if( board[r][c] == '.'){
                    for( char i = '1'; i <= '9'; i++) {
                        if( is_valid(board, r, c, i) ){
                            board[r][c] = i;
                            if(solveSudoku(board)){
                                return true;
                            }
                            board[r][c] = '.';
                        }
                    }
                    return false;
                }
            }
        }
        return true;
    }
    bool is_valid(const vector<vector<char>>& board, int r, int c, char val){
        for(int i = 0; i < 9; i++){
            if(board[r][i] == val ) return false;
            if(board[i][c] == val ) return false;

            int boxRow = 3 * (r / 3) + (i/3);
            int boxCol = 3 * (c / 3) + (i%3);
            if( board[boxRow][boxCol] == val) return false;
        }
        return true;
    }                         
};

public class Solution {
    public void SolveSudoku(char[][] board) {
        Solve(ref board);
    }
    public bool Solve(ref char[][] board) {
        for(int r = 0; r < 9; r++){
            for(int c = 0; c < 9; c++){
                if( board[r][c] == '.'){
                    for(char val = '1'; val <= '9'; val++){
                        if( check(ref board,r,c,val) ){
                            board[r][c] = val;
                            if( Solve(ref board) ){
                                return true;
                            }
                            board[r][c] = '.';
                        }
                    }
                    return false;
                }
            }
        }
        return true;
    }
    bool check(ref char[][] board, int r, int c, char val ){
        for(int i = 0; i < 9; i++){
            if(board[r][i] == val) return false;
            if(board[i][c] == val) return false;
            int rowIdx = 3 * ( r / 3) + (i/3);
            int colIdx = 3 * ( c / 3) + (i%3);
            if(board[rowIdx][colIdx] == val) return false;
        }
        return true;
    }
}
