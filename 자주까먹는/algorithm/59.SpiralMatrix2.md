# 59. Spiral Matrix II

Given a positive integer n, generate an n x n matrix filled with elements from 1 to n2 in spiral order.

 

Example 1:


Input: n = 3
Output: [[1,2,3],[8,9,4],[7,6,5]]
Example 2:

Input: n = 1
Output: [[1]]
 

Constraints:

1 <= n <= 20

## My Solution

```cpp
class Solution {
public:
    vector<vector<int>> generateMatrix(int n) {
        vector<vector<int>> result(n, vector<int>(n));
        int left{}, top{};
        int right{n - 1}, bottom{n-1};
        int current {1};
        while(left <= right && top <= bottom){
            for( int i = left; i <= right; i++){
                result[top][i] = current++;
            }
            top++;
            for(int i = top; i <= bottom; i++){
                result[i][right] = current++;
            }
            right--;
            for(int i = right; i >= left; i--){
                result[bottom][i] = current++;
            }
            bottom--;
            for(int i = bottom; i >= top; i--){
                result[i][left] = current++;
            }
            left++;
        }
        return result;
    }
};
```

### 정사각행렬이 아닌경우 

```cpp
#include <iostream>
#include <vector>

using namespace std;

class Solution {
public:
    vector<vector<int>> generateMatrix(int n) {
        // n x n 2차원 벡터를 0으로 초기화하여 생성
        vector<vector<int>> matrix(n, vector<int>(n, 0));
        
        // 4면의 경계선 인덱스 초기화
        int top = 0, bottom = n - 1;
        int left = 0, right = n - 1;
        
        int current = 1; // 1부터 시작
        
        // 경계선이 서로 교차하기 전까지 반복
        while (left <= right && top <= bottom) {
            
            // 1. 위쪽(Top) 테두리: 왼쪽에서 오른쪽으로
            for (int i = left; i <= right; ++i) {
                matrix[top][i] = current++;
            }
            top++; // 위쪽 껍질을 채웠으므로 경계를 아래로 1칸 내림
            
            // 2. 오른쪽(Right) 테두리: 위에서 아래로
            for (int i = top; i <= bottom; ++i) {
                matrix[i][right] = current++;
            }
            right--; // 오른쪽 껍질을 채웠으므로 경계를 왼쪽으로 1칸 당김
            
            // 3. 아래쪽(Bottom) 테두리: 오른쪽에서 왼쪽으로
            if (top <= bottom) { // 행 경계가 유효한지 안전 검사
                for (int i = right; i >= left; --i) {
                    matrix[bottom][i] = current++;
                }
                bottom--; // 아래쪽 껍질을 채웠으므로 경계를 위로 1칸 올림
            }
            
            // 4. 왼쪽(Left) 테두리: 아래에서 위로
            if (left <= right) { // 열 경계가 유효한지 안전 검사
                for (int i = bottom; i >= top; --i) {
                    matrix[i][left] = current++;
                }
                left++; // 왼쪽 껍질을 채웠으므로 경계를 오른쪽으로 1칸 밈
            }
        }
        
        return matrix;
    }
};

int main() {
    Solution sol;
    vector<vector<int>> result = sol.generateMatrix(3);
    
    // 결과 출력
    for (const auto& row : result) {
        for (int val : row) {
            cout << val << "\t";
        }
        cout << "\n";
    }
    return 0;
}
```
